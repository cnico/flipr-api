Commandes utiles :

#Pour mettre à jour les dépendances
poetry update

# Pour installer le hook de pre-commit :
poetry run pre-commit install

#Pour lancer black
poetry run black .

#Pour lancer ruff
poetry run ruff format


#Pour lancer les tests
poetry run pytest -s

clear ; poetry run pytest -s --cov=src tests/test_integrations.py

#Pour tester la doc
poetry run mkdocs serve


#Pour publier sur codecov :
poetry run pytest --cov=src --cov-report=xml ; bash <(curl -s https://codecov.io/bash ) -t 68fa5301-c989-4415-83e5-5c43c1bb62fe


#Publication pypi :
poetry build
twine upload dist/flipr_api-1.6.0-py3-none-any.whl --verbose
